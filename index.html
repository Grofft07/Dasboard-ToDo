// ============================================================ // APPS SCRIPT — TO DO LIST KERJA + EVALUASI MINGGUAN // Paste di Google Apps Script (Extensions -> Apps Script dari spreadsheet // kamu) -> Deploy -> New deployment -> Web app // Execute as : Me // Who has access: Anyone // Salin URL hasil deploy ke kolom "URL Web App Apps Script" di dashboard. // Sheet "ToDo", "Evaluasi Mingguan", "Karyawan Baru", "Log Karyawan", dan // "Checklist Masalah" akan dibuat otomatis di spreadsheet // yang sama. // ============================================================ var SHEET_NAME = 'ToDo'; var HEADERS = ['ID','TanggalDibuat','Kategori','Prioritas','Tugas','Deadline','Status','Catatan']; var EVAL_SHEET_NAME = 'Evaluasi Mingguan'; var EVAL_HEADERS = ['ID','Tanggal','Catatan']; var EMP_SHEET_NAME = 'Karyawan Baru'; var EMP_HEADERS = ['ID','Nama','Posisi','Divisi','TanggalMulai','Status','Catatan']; var LOG_SHEET_NAME = 'Log Karyawan'; var LOG_HEADERS = ['ID','KaryawanID','Tanggal','Tugas','Alasan','Progress','Catatan']; var PROBLEM_SHEET_NAME = 'Checklist Masalah'; var PROBLEM_HEADERS = ['ID','Tanggal','Section','Item','Catatan','Status']; var KEU_SHEET_NAME = 'Uang Harian'; var KEU_HEADERS = ['ID','Tanggal','Tipe','Kategori','Jumlah','Keterangan']; function getSheet_(){ var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(SHEET_NAME); if (!sh){ sh = ss.insertSheet(SHEET_NAME); sh.appendRow(HEADERS); sh.setFrozenRows(1); } return sh; } function getEvalSheet_(){ var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(EVAL_SHEET_NAME); if (!sh){ sh = ss.insertSheet(EVAL_SHEET_NAME); sh.appendRow(EVAL_HEADERS); sh.setFrozenRows(1); } return sh; } function getEmpSheet_(){ var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(EMP_SHEET_NAME); if (!sh){ sh = ss.insertSheet(EMP_SHEET_NAME); sh.appendRow(EMP_HEADERS); sh.setFrozenRows(1); } return sh; } function getLogSheet_(){ var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(LOG_SHEET_NAME); if (!sh){ sh = ss.insertSheet(LOG_SHEET_NAME); sh.appendRow(LOG_HEADERS); sh.setFrozenRows(1); } return sh; } function getProblemSheet_(){ var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(PROBLEM_SHEET_NAME); if (!sh){ sh = ss.insertSheet(PROBLEM_SHEET_NAME); sh.appendRow(PROBLEM_HEADERS); sh.setFrozenRows(1); } return sh; } function getKeuSheet_(){ var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(KEU_SHEET_NAME); if (!sh){ sh = ss.insertSheet(KEU_SHEET_NAME); sh.appendRow(KEU_HEADERS); sh.setFrozenRows(1); } return sh; } function doGet(e){ var sh = getSheet_(); var shEval = getEvalSheet_(); var shEmp = getEmpSheet_(); var shLog = getLogSheet_(); var shProblem = getProblemSheet_(); var shKeu = getKeuSheet_(); return jsonOutput({ tasks: sheetToObjects_(sh), evals: sheetToObjects_(shEval), employees: sheetToObjects_(shEmp), logs: sheetToObjects_(shLog), problems: sheetToObjects_(shProblem), keuangan: sheetToObjects_(shKeu) }); } function doPost(e){ var data = JSON.parse(e.postData.contents); var lock = LockService.getScriptLock(); try{ lock.waitLock(30000); } catch(lockErr){ return jsonOutput({ status:'error', message:'Server sibuk, coba lagi.' }); } try{ if (data.sheet === 'eval'){ var shEval = getEvalSheet_(); if (data.type === 'upsert') upsertEval_(shEval, data); else if (data.type === 'delete') deleteEval_(shEval, data); } else if (data.sheet === 'karyawan'){ var shEmp = getEmpSheet_(); if (data.type === 'upsert') upsertEmp_(shEmp, data); else if (data.type === 'delete') deleteEmp_(shEmp, data); } else if (data.sheet === 'logkaryawan'){ var shLog = getLogSheet_(); if (data.type === 'upsert') upsertLog_(shLog, data); else if (data.type === 'delete') deleteLog_(shLog, data); } else if (data.sheet === 'checklistmasalah'){ var shProblem = getProblemSheet_(); if (data.type === 'upsert') upsertProblem_(shProblem, data); else if (data.type === 'delete') deleteProblem_(shProblem, data); } else if (data.sheet === 'keuangan'){ var shKeu = getKeuSheet_(); if (data.type === 'upsert') upsertKeu_(shKeu, data); else if (data.type === 'delete') deleteKeu_(shKeu, data); } else { var sh = getSheet_(); if (data.type === 'upsert') upsertTask_(sh, data); else if (data.type === 'delete') deleteTask_(sh, data); } } finally { lock.releaseLock(); } return jsonOutput({ status:'ok' }); } function jsonOutput(obj){ return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function formatCellDate_(v){ if (Object.prototype.toString.call(v) === '[object Date]'){ return Utilities.formatDate(v, SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone(), 'yyyy-MM-dd'); } return String(v || '').trim(); } var DATE_FIELDS_ = { deadline:1, tanggaldibuat:1, tanggal:1, tenggat:1, tanggalmulai:1 }; function sheetToObjects_(sheet){ var rows = sheet.getDataRange().getValues(); }); if (isEmpty) continue; var obj = {}; headers.forEach(function(h, idx){ var v = row[idx]; obj[h] = DATE_FIELDS_[h] ? formatCellDate_(v) : v; }); out.push(obj); } return out; } // Upsert berdasarkan ID (baris baru kalau ID belum ada, update kalau sudah ada) function upsertTask_(sh, data){ var rows = sh.getDataRange().getValues(); return; } } sh.appendRow(rowValues); } function deleteTask_(sh, data){ var rows = sh.getDataRange().getValues(); var id = String(data.id).trim(); for (var i = 1; i < rows.length; i++){ if (String(rows[i][0]).trim() === id){ sh.deleteRow(i + 1); return; } } } // Upsert catatan mingguan berdasarkan ID function upsertEval_(sh, data){ var rows = sh.getDataRange().getValues(); var id = String(data.id).trim(); var rowValues = [data.id, data.tanggal, data.what]; for (var i = 1; i < rows.length; i++){ if (String(rows[i][0]).trim() === id){ sh.getRange(i + 1, 1, 1, LOG_HEADERS.length).setValues([rowValues]); return; } } sh.appendRow(rowValues); } function deleteLog_(sh, data){ var rows = sh.getDataRange().getValues(); var id = String(data.id).trim(); var rowValues = [data.id, data.tanggal, data.section, data.item, data.catatan, data.status]; for (var i = 1; i < rows.length; i++){ if (String(rows[i][0]).trim() === id){ sh.getRange(i + 1, 1, 1, PROBLEM_HEADERS.length).setValues([rowValues]); return; } } } // Upsert transaksi uang harian berdasarkan ID function upsertKeu_(sh, data){ var rows = sh.getDataRange().getValues(); var id = String(data.id).trim(); var rowValues = [data.id, data.tanggal, data.tipe, data.kategori, data.jumlah, data.keterangan]; for (var i = 1; i < rows.length; i++){ if (String(rows[i][0]).trim() === id){ sh.getRange(i + 1, 1, 1, KEU_HEADERS.length).setValues([rowValues]); return; } } sh.appendRow(rowValues); } function deleteKeu_(sh, data){ var rows = sh.getDataRange().getValues();
